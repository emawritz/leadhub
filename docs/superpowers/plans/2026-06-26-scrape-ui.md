# Scrape-on-demand desde la UI — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agregar una superficie "Buscar" a LeadHub: desde la web se lanza una búsqueda de Google Maps, un worker corre el scraper con progreso en vivo, y al terminar los leads entran solos a la base (visibles en Explorar).

**Architecture:** El scraper (`@leadhub/scraper`) gana un flag `--progress-json` que emite NDJSON de progreso a stdout. La API (`@leadhub/api`) gana una tabla `scrape_jobs`, endpoints `/scrape`, y un worker in-process (concurrencia 1) que spawnea el scraper como subproceso, parsea su progreso a la DB, y al terminar hace `upsertLeads`. La web (`@leadhub/web`) gana una página `/buscar` con formulario + lista de jobs con polling.

**Tech Stack:** Node 20 + TS strict (ESM), pnpm workspaces, Fastify, Drizzle + PostgreSQL, Zod, `node:child_process` + `node:readline`, React + Vite + TanStack Query, Vitest.

## Global Constraints

- TypeScript **strict**; ESM (`type: module`), imports con extensión `.js`.
- Gestor **pnpm**; paquetes existentes: `@leadhub/core`, `@leadhub/scraper`, `@leadhub/api`, `@leadhub/ingest`, `@leadhub/web`.
- Envelope de API: `{ ok, data, error }`; validación **Zod**; `requireAuth` (JWT) en todos los endpoints nuevos.
- Postgres en host port **5544** (`DATABASE_URL`/`DATABASE_URL_TEST` en apps/api/.env). La API se corre desde la **raíz del repo** (para que `output/` y `apps/scraper/dist/cli.js` resuelvan).
- Puerto de la API en esta máquina: **4501** (3001 está ocupado por un servicio PM2). Web con `VITE_API_PROXY_TARGET=http://localhost:4501`.
- Scraping: **concurrencia 1** (serializa; anti-bot + un solo perfil de Google). **Headless** por defecto.
- Progreso: NDJSON en **stdout** del scraper; logs humanos no deben romper el parseo (se desactiva la barra cuando hay `onProgress`).
- No romper los tests existentes: core 28, scraper 20, api 10, ingest 2, web 2 + E2E.

```typescript
import type { ProgressEvent } from '@leadhub/core';

export function parseProgressLine(line: string): ProgressEvent | null {
  const t = line.trim();
  if (!t.startsWith('{')) return null;
  try {
    const obj = JSON.parse(t) as Partial<ProgressEvent>;
    return typeof obj.phase === 'string' ? (obj as ProgressEvent) : null;
  } catch {
    return null;
  }
}
```

```typescript
import { eq } from 'drizzle-orm';
import type { NodePgDatabase } from 'drizzle-orm/node-postgres';
import type { ProgressEvent } from '@leadhub/core';
import { scrapeJobs } from '../db/schema.js';

export type ScrapeJobRow = typeof scrapeJobs.$inferSelect;

export type RunJob = (
  job: ScrapeJobRow,
  hooks: { onProgress: (ev: ProgressEvent) => Promise<void> },
) => Promise<{ leadsFound: number; inserted: number; updated: number }>;

export function makeScrapeWorker(db: NodePgDatabase, runJob: RunJob) {
  const queue: string[] = [];
  let processing = false;

  async function processNext(): Promise<void> {
    if (processing || queue.length === 0) return;
    processing = true;
    const id = queue.shift()!;
    try {
      const [job] = await db.select().from(scrapeJobs).where(eq(scrapeJobs.id, id)).limit(1);
      if (!job) return;
      await db.update(scrapeJobs).set({ status: 'running', startedAt: new Date() }).where(eq(scrapeJobs.id, id));
      const onProgress = async (ev: ProgressEvent) => {
        await db.update(scrapeJobs).set({
          phase: ev.phase,
          ...(ev.done != null ? { done: ev.done } : {}),
          ...(ev.total != null ? { total: ev.total } : {}),
        }).where(eq(scrapeJobs.id, id));
      };
      try {
        const r = await runJob(job, { onProgress });
        await db.update(scrapeJobs).set({
          status: 'done', leadsFound: r.leadsFound, inserted: r.inserted, updated: r.updated,
          finishedAt: new Date(),
        }).where(eq(scrapeJobs.id, id));
      } catch (e) {
        await db.update(scrapeJobs).set({
          status: 'error', error: (e as Error).message.slice(0, 500), finishedAt: new Date(),
        }).where(eq(scrapeJobs.id, id));
      }
    } finally {
      processing = false;
      void processNext();
    }
  }

  return {
    enqueue(id: string): void {
      queue.push(id);
      void processNext();
    },
  };
}
```

```typescript
import type { FastifyInstance } from 'fastify';
import { desc, eq } from 'drizzle-orm';
import type { NodePgDatabase } from 'drizzle-orm/node-postgres';
import { scrapeInputSchema } from './input.js';
import { scrapeJobs } from '../db/schema.js';
import { requireAuth } from '../auth/require-auth.js';

interface Worker { enqueue(id: string): void; }

export function scrapeRoutes(app: FastifyInstance, db: NodePgDatabase, worker: Worker) {
  app.post('/scrape', { preHandler: [requireAuth] }, async (req, reply) => {
    const parsed = scrapeInputSchema.safeParse(req.body);
    if (!parsed.success) return reply.code(400).send({ ok: false, error: 'Datos inválidos' });
    const { query, location, max, enrich, excludeChains } = parsed.data;
    const accountId = (req.user as { sub: string }).sub;
    const [job] = await db.insert(scrapeJobs)
      .values({ accountId, query, location, max, enrich, excludeChains }).returning();
    worker.enqueue(job.id);
    return { ok: true, data: job };
  });

  app.get('/scrape', { preHandler: [requireAuth] }, async () => {
    const rows = await db.select().from(scrapeJobs).orderBy(desc(scrapeJobs.createdAt)).limit(50);
    return { ok: true, data: rows };
  });

  app.get('/scrape/:id', { preHandler: [requireAuth] }, async (req, reply) => {
    const { id } = req.params as { id: string };
    const [row] = await db.select().from(scrapeJobs).where(eq(scrapeJobs.id, id)).limit(1);
    if (!row) return reply.code(404).send({ ok: false, error: 'No encontrado' });
    return { ok: true, data: row };
  });
}
```

```typescript
import { spawn } from 'node:child_process';
import { createInterface } from 'node:readline';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { eq } from 'drizzle-orm';
import type { NodePgDatabase } from 'drizzle-orm/node-postgres';
import type { Lead } from '@leadhub/core';
import { upsertLeads } from '@leadhub/ingest';
import { parseProgressLine } from './progress.js';
import { scrapeJobs } from '../db/schema.js';
import type { RunJob } from './worker.js';

const REPO_ROOT = process.cwd();
const SCRAPER_CLI = path.resolve(REPO_ROOT, 'apps/scraper/dist/cli.js');

export function spawnScraperAndIngest(db: NodePgDatabase): RunJob {
  return async (job, hooks) => {
    const outputPath = `output/jobs/${job.id}.json`;
    await db.update(scrapeJobs).set({ outputPath }).where(eq(scrapeJobs.id, job.id));

    const args = [
      SCRAPER_CLI, 'search',
      '--query', job.query, '--location', job.location, '--max', String(job.max),
      '--progress-json', '--output', outputPath,
    ];
    if (job.enrich) args.push('--enrich');
    if (job.excludeChains) args.push('--exclude-chains');

    const child = spawn('node', args, { cwd: REPO_ROOT });
    let stderr = '';
    const rl = createInterface({ input: child.stdout });
    rl.on('line', (line) => {
      const ev = parseProgressLine(line);
      if (ev) void hooks.onProgress(ev);
    });
    child.stderr.on('data', (d) => { stderr += String(d); });

    const code: number = await new Promise((resolve) => child.on('close', (c) => resolve(c ?? 1)));
    if (code !== 0) {
      const captcha = /captcha|unusual traffic/i.test(stderr);
      throw new Error(captcha
        ? 'Google pidió verificación — corré el scraper con --headed una vez para reautenticar el perfil.'
        : `El scraper falló (código ${code}). ${stderr.slice(-300)}`);
    }

    const leads = JSON.parse(await readFile(path.join(REPO_ROOT, outputPath), 'utf8')) as Lead[];
    const r = await upsertLeads(db, leads);
    return { leadsFound: leads.length, inserted: r.inserted, updated: r.updated };
  };
}
```

```typescript
export function jobLabel(job: ScrapeJob): string {
```
