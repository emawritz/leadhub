# CRM UX del tablero (Fase 2) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Elevar el Kanban y el drawer del CRM de "funciona" a una UI intencional B2B (jerarquía, color semántico por etapa, estados diseñados, drag pulido), sin cambiar el modelo de datos.

**Architecture:** Se agrega un único punto de tokens de color/estilo por etapa y por tipo de actividad en `packages/core/src/stages.ts` (consumido por web vía el subpath `@leadhub/core/stages`, con **valores hex** aplicados por `style` inline para no depender del purge de Tailwind). El board de la API gana un campo `proximoVence` (fecha de la tarea pendiente más próxima) para poder mostrar vencimiento en la tarjeta. Los componentes web (`KanbanCard`, `KanbanBoard`, `LeadDrawer`, `Tareas`) se reescriben sobre esos tokens.

**Tech Stack:** React 18 + Vite + Tailwind + @dnd-kit; Fastify + Drizzle (una subquery correlacionada nueva en el board); TypeScript ESM.

## Global Constraints

- **SIN tests automatizados** (decisión del proyecto). Verificación por **typecheck/build verde** + **smoke visual** (capturas a 1440/1024/768 y chequeo de teclado + reduced-motion). No agregar archivos de test. No romper los tests existentes del monorepo.
- **Value-imports de `@leadhub/core` desde web SOLO por el subpath `@leadhub/core/stages`** (el barrel raíz rompe el build de Vite).
- **Colores por etapa/actividad como valores hex aplicados con `style={{...}}` inline**, no como clases Tailwind dinámicas (`bg-${x}` se purga). Los tokens viven una sola vez en `stages.ts` (DRY).
- Animaciones **solo en propiedades compositor-friendly** (`transform`, `opacity`, `box-shadow`); respetar `prefers-reduced-motion` (sin animación de "levantar" si el usuario lo pide).
- Se conserva el **drag optimista con rollback** ya existente (no cambiar la lógica de la mutación `move`).
- Envelope de API `{ ok, data, error }`; rutas con `requireAuth`.
- Dirección de estilo: **B2B claro y denso, con carácter** (no dark por defecto, no template genérico). Aplicar la checklist de calidad: jerarquía por escala, ritmo de espaciado intencional, profundidad por sombra/superficie, color semántico, estados hover/focus/active diseñados.
- Commits en español, formato convencional, usando EXACTAMENTE:
  `git -c user.name="Emanuel Cejas" -c user.email="emanuel.cejas.ar@gmail.com" commit -m "..."`

```tsx
import { useDraggable } from '@dnd-kit/core';
import type { PipelineCard } from '../api.js';

/** Color estable derivado de un string (para el avatar del dueño). */
function avatarColor(seed: string): string {
  const palette = ['#0ea5e9', '#8b5cf6', '#f59e0b', '#10b981', '#ef4444', '#ec4899', '#14b8a6', '#6366f1'];
  let h = 0;
  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) >>> 0;
  return palette[h % palette.length];
}

function initials(name: string | null): string {
  if (!name) return '–';
  const parts = name.trim().split(/\s+/).slice(0, 2);
  return parts.map((p) => p[0]?.toUpperCase() ?? '').join('') || '–';
}

export default function KanbanCard({ card, onOpen }: { card: PipelineCard; onOpen: (id: string) => void }) {
  const { attributes, listeners, setNodeRef, transform, isDragging } = useDraggable({ id: card.id });
  const style = transform ? { transform: `translate(${transform.x}px, ${transform.y}px)` } : undefined;
  const vence = card.proximoVence ? new Date(card.proximoVence) : null;
  const vencida = vence != null && vence.getTime() < Date.now();
  const ownerBg = card.ownerNombre ? avatarColor(card.ownerId ?? card.ownerNombre) : '#cbd5e1';

  return (
    <div
      ref={setNodeRef}
      style={style}
      {...listeners}
      {...attributes}
      onClick={() => onOpen(card.id)}
      className={`group bg-white border border-slate-200 rounded-lg p-2.5 mb-2 cursor-grab shadow-sm transition-shadow hover:shadow-md hover:border-slate-300 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-slate-400 ${isDragging ? 'opacity-50' : ''}`}
    >
      <div className="font-semibold text-sm text-slate-800 truncate">{card.nombre}</div>
      <div className="mt-1 flex items-center gap-1.5 text-xs text-slate-500">
        {card.categoria && <span className="bg-slate-100 text-slate-600 rounded px-1.5 py-0.5 text-[11px] truncate max-w-[8rem]">{card.categoria}</span>}
        {card.localidad && <span className="truncate">📍 {card.localidad}</span>}
      </div>
      <div className="mt-2 flex items-center justify-between gap-2">
        <div className="flex items-center gap-1.5">
          <span
            className="inline-flex items-center justify-center w-5 h-5 rounded-full text-[10px] font-semibold text-white"
            style={{ backgroundColor: ownerBg }}
            title={card.ownerNombre ?? 'sin dueño'}
          >
            {initials(card.ownerNombre)}
          </span>
        </div>
        <div className="flex items-center gap-1.5">
          {vence && (
            <span className={`text-[11px] px-1.5 py-0.5 rounded ${vencida ? 'bg-red-50 text-red-600' : 'bg-slate-100 text-slate-500'}`}>
              ⏰ {vence.toLocaleDateString()}
            </span>
          )}
          {card.pendientes > 0 && (
            <span className="text-[11px] px-1.5 py-0.5 rounded bg-amber-100 text-amber-700 font-medium">{card.pendientes}</span>
          )}
        </div>
      </div>
    </div>
  );
}
```

```tsx
import { useState } from 'react';
import { DndContext, DragOverlay, useDroppable, type DragEndEvent, type DragStartEvent, PointerSensor, useSensor, useSensors } from '@dnd-kit/core';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
// Import directo del subpath para evitar que Vite empaquete el resto de @leadhub/core.
import { STAGES, CLOSED_STAGES, stageColor } from '@leadhub/core/stages';
import { getPipeline, updatePipeline, type PipelineCard } from '../api.js';
import KanbanCard from './KanbanCard.js';

function useReducedMotion(): boolean {
  return typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}

function Column({ stageId, label, cards, onOpen }: { stageId: string; label: string; cards: PipelineCard[]; onOpen: (id: string) => void }) {
  const { setNodeRef, isOver } = useDroppable({ id: stageId });
  const color = stageColor(stageId);
  const closed = (CLOSED_STAGES as readonly string[]).includes(stageId);
  return (
    <div
      ref={setNodeRef}
      style={{ borderTop: `3px solid ${color}`, boxShadow: isOver ? `inset 0 0 0 2px ${color}` : undefined }}
      className={`w-64 shrink-0 rounded-lg p-2 transition-colors ${isOver ? 'bg-slate-100' : 'bg-slate-50'}`}
    >
      <div className="flex items-center justify-between px-1 pb-2">
        <span className="text-sm font-semibold text-slate-700 flex items-center gap-1">
          <span className="inline-block w-2 h-2 rounded-full" style={{ backgroundColor: color }} />
          {closed ? `✓ ${label}` : label}
        </span>
        <span className="text-xs text-slate-400">{cards.length}</span>
      </div>
      {cards.length === 0
        ? <div className="text-xs text-slate-400 border border-dashed border-slate-200 rounded p-3 text-center">Vacío</div>
        : cards.map((c) => <KanbanCard key={c.id} card={c} onOpen={onOpen} />)}
    </div>
  );
}

export default function KanbanBoard({ onOpen }: { onOpen: (id: string) => void }) {
  const qc = useQueryClient();
  const reduced = useReducedMotion();
  const [activeId, setActiveId] = useState<string | null>(null);
  const pipeline = useQuery({ queryKey: ['pipeline'], queryFn: getPipeline });
  const move = useMutation({
    mutationFn: ({ id, stage }: { id: string; stage: string }) => updatePipeline(id, { stage }),
    onMutate: async ({ id, stage }) => {
      await qc.cancelQueries({ queryKey: ['pipeline'] });
      const prev = qc.getQueryData<PipelineCard[]>(['pipeline']);
      qc.setQueryData<PipelineCard[]>(['pipeline'], (old) => (old ?? []).map((c) => (c.id === id ? { ...c, stage } : c)));
      return { prev };
    },
    onError: (_e, _v, ctx) => { if (ctx?.prev) qc.setQueryData(['pipeline'], ctx.prev); },
    onSettled: () => qc.invalidateQueries({ queryKey: ['pipeline'] }),
  });
  const sensors = useSensors(useSensor(PointerSensor, { activationConstraint: { distance: 5 } }));
  const cards = pipeline.data ?? [];
  const activeCard = activeId ? cards.find((c) => c.id === activeId) ?? null : null;

  function onDragStart(e: DragStartEvent) { setActiveId(String(e.active.id)); }
  function onDragEnd(e: DragEndEvent) {
    setActiveId(null);
    const id = String(e.active.id);
    const stage = e.over ? String(e.over.id) : null;
    if (!stage) return;
    const card = cards.find((c) => c.id === id);
    if (card && card.stage !== stage) move.mutate({ id, stage });
  }

  return (
    <DndContext sensors={sensors} onDragStart={onDragStart} onDragEnd={onDragEnd}>
      <div className="flex gap-3 overflow-x-auto p-4">
        {STAGES.map((s) => (
          <Column key={s.id} stageId={s.id} label={s.label} cards={cards.filter((c) => c.stage === s.id)} onOpen={onOpen} />
        ))}
      </div>
      <DragOverlay dropAnimation={reduced ? null : undefined}>
        {activeCard
          ? <div className={reduced ? '' : 'shadow-xl rounded-lg'}><KanbanCard card={activeCard} onOpen={() => {}} /></div>
          : null}
      </DragOverlay>
    </DndContext>
  );
}
```

  const [vence, setVence] = useState('');
  const actMut = useMutation({ mutationFn: () => addActivity(pipelineId, { tipo, cuerpo }), onSuccess: () => { setCuerpo(''); inval(); } });
  const taskMut = useMutation({ mutationFn: () => addTask(pipelineId, { titulo, vence: vence ? new Date(vence).toISOString() : undefined }), onSuccess: () => { setTitulo(''); setVence(''); inval(); } });
  const toggleMut = useMutation({ mutationFn: (t: { id: string; hecha: boolean }) => toggleTask(t.id, t.hecha), onSuccess: inval });
  const stageMut = useMutation({ mutationFn: (stage: string) => updatePipeline(pipelineId, { stage }), onSuccess: inval });
  const delMut = useMutation({ mutationFn: () => removeFromPipeline(pipelineId), onSuccess: () => { onClose(); qc.invalidateQueries({ queryKey: ['pipeline'] }); } });

  const d = detail.data;
  const inputCls = 'border border-slate-300 rounded px-2 py-1 text-sm focus:outline-none focus:ring-2 focus:ring-slate-300';
  return (
    <div className="fixed inset-0 bg-black/30 flex justify-end z-50" onClick={onClose}>
      <div className="w-[28rem] bg-white h-full overflow-y-auto p-5" onClick={(e) => e.stopPropagation()}>
        {!d ? <p className="text-slate-400">Cargando…</p> : (
          <>
            <div className="flex justify-between items-start">
              <div>
                <h2 className="font-bold text-lg text-slate-800">{d.lead.nombreFantasia}</h2>
                <p className="text-sm text-slate-500">{d.lead.categoria} · {d.lead.localidad}</p>
                <p className="text-sm text-slate-600">{d.lead.telefono} · {d.lead.email}</p>
              </div>
              <button onClick={onClose} className="text-slate-400 hover:text-slate-600 text-lg leading-none">✕</button>
            </div>

            <div className="mt-4 flex items-center gap-2">
              <span className="inline-flex items-center gap-1.5 text-sm font-medium">
                <span className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: stageColor(d.pipeline.stage) }} />
                Etapa
              </span>
              <select value={d.pipeline.stage} onChange={(e) => stageMut.mutate(e.target.value)} className={inputCls}>
                {STAGES.map((s) => <option key={s.id} value={s.id}>{s.label}</option>)}
              </select>
              <button onClick={() => delMut.mutate()} className="ml-auto text-sm text-red-600 hover:underline">Sacar</button>
            </div>

            <h3 className="font-semibold text-slate-700 mt-6 mb-2">Tareas</h3>
            <div className="flex gap-1.5 mb-2">
              <input value={titulo} onChange={(e) => setTitulo(e.target.value)} placeholder="Nueva tarea" className={`${inputCls} flex-1`} />
              <input type="datetime-local" value={vence} onChange={(e) => setVence(e.target.value)} className={inputCls} />
              <button disabled={!titulo || taskMut.isPending} onClick={() => taskMut.mutate()} className="bg-slate-800 text-white px-2.5 rounded text-sm disabled:opacity-40">+</button>
            </div>
            <ul className="text-sm space-y-1">
              {d.tasks.map((t) => {
                const venc = t.vence ? new Date(t.vence) : null;
                const vencida = venc != null && venc.getTime() < Date.now() && !t.hecha;
                return (
                  <li key={t.id} className="flex items-center gap-2">
                    <input type="checkbox" checked={t.hecha} onChange={(e) => toggleMut.mutate({ id: t.id, hecha: e.target.checked })} />
                    <span className={`flex-1 ${t.hecha ? 'line-through text-slate-400' : 'text-slate-700'}`}>{t.titulo}</span>
                    {venc && <span className={`text-xs ${vencida ? 'text-red-600 font-medium' : 'text-slate-400'}`}>{venc.toLocaleDateString()}</span>}
                  </li>
                );
              })}
              {d.tasks.length === 0 && <li className="text-slate-400 text-xs">Sin tareas</li>}
            </ul>

            <h3 className="font-semibold text-slate-700 mt-6 mb-2">Actividad</h3>
            <div className="flex gap-1.5 mb-3">
              <select value={tipo} onChange={(e) => setTipo(e.target.value)} className={inputCls}>
                {USER_ACTIVITY_TYPES.map((t) => <option key={t} value={t}>{t}</option>)}
              </select>
              <input value={cuerpo} onChange={(e) => setCuerpo(e.target.value)} placeholder="Registrar…" className={`${inputCls} flex-1`} />
              <button disabled={!cuerpo || actMut.isPending} onClick={() => actMut.mutate()} className="bg-slate-800 text-white px-2.5 rounded text-sm disabled:opacity-40">+</button>
            </div>
            <ul className="space-y-2">
              {d.activities.map((a) => {
                const meta = (ACTIVITY_META as Record<string, { icon: string; color: string }>)[a.tipo] ?? NEUTRAL;
                return (
                  <li key={a.id} className="pl-3 py-0.5" style={{ borderLeft: `3px solid ${meta.color}` }}>
                    <div className="text-sm text-slate-700"><span className="mr-1">{meta.icon}</span><span className="text-xs uppercase tracking-wide text-slate-400 mr-1">{a.tipo}</span>{a.cuerpo}</div>
                    <div className="text-xs text-slate-400">{new Date(a.createdAt).toLocaleString()}</div>
                  </li>
                );
              })}
              {d.activities.length === 0 && <li className="text-slate-400 text-xs">Sin actividad todavía</li>}
            </ul>
          </>
        )}
      </div>
    </div>
  );
}
